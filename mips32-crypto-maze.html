uint32_t rotl(uint32_t v, unsigned r) { r &= 31; return (v << r) | (v >> ((32 - r) & 31)); } uint32_t rotr(uint32_t v, unsigned r) { return rotl(v, (32 - r) & 31); } uint32_t encrypt(uint32_t x, uint32_t y, uint32_t id) { return rotl(((x << 16) | y) ^ id, id % 32); } void decrypt(uint32_t c, uint32_t id, uint32_t *x, uint32_t *y) { uint32_t t = rotr(c, id % 32) ^ id; *x = (t >> 16) & 0xFFFF; *y = t & 0xFFFF; }
